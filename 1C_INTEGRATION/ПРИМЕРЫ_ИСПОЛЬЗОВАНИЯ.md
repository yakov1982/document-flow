## Отправка документа на согласование

### Пример 1: Отправка счета на оплату

```1c
Процедура ОтправитьСчетВСЭД(СчетСсылка)
    
    // Создание обработки интеграции
    Обработка = ВнешниеОбработки.Создать("ИнтеграцияДМС");
    Обработка.УстановитьПараметрыПодключения(
        Константы.АдресСерверДМС.Получить(),
        Константы.ЛогинДМС.Получить(),
        Константы.ПарольДМС.Получить()
    );
    
    // Формирование заголовка документа
    Заголовок = СтрШаблон("Счет на оплату №%1 от %2",
        СчетСсылка.Номер,
        Формат(СчетСсылка.Дата, "ДФ=dd.MM.yyyy"));
    
    // Определение согласующих
    МассивСогласующих = Новый Массив;
    МассивСогласующих.Добавить("manager");
    МассивСогласующих.Добавить("director");
    
    // Отправка документа
    Результат = Обработка.ОтправитьДокументВСЭД(
        СчетСсылка,
        Заголовок,
        "invoice",
        МассивСогласующих,
        Истина  // Прикрепить печатную форму
    );
    
    Если Результат.Успех Тогда
        
        // Сохранение ID документа в СЭД
        // Для этого создайте реквизит "ИдентификаторВСЭД" у документа
        СчетОбъект = СчетСсылка.ПолучитьОбъект();
        СчетОбъект.ИдентификаторВСЭД = Результат.ИдентификаторДокумента;
        СчетОбъект.СтатусСогласования = Перечисления.СтатусыСогласования.НаСогласовании;
        СчетОбъект.Записать();
        
        Сообщить(СтрШаблон("Счет отправлен в СЭД. ID: %1", Результат.ИдентификаторДокумента));
        
    Иначе
        
        Сообщить("Ошибка отправки: " + Результат.ТекстОшибки);
        
    КонецЕсли;
    
КонецПроцедуры
```

### Пример 2: Отправка договора с контрагентом

```1c
Процедура ОтправитьДоговорВСЭД(ДоговорСсылка, Знач МассивСогласующих = Неопределено)
    
    Обработка = ВнешниеОбработки.Создать("ИнтеграцияДМС");
    Обработка.УстановитьПараметрыПодключения(
        Константы.АдресСерверДМС.Получить(),
        Константы.ЛогинДМС.Получить(),
        Константы.ПарольДМС.Получить()
    );
    
    // Формирование заголовка
    Заголовок = СтрШаблон("Договор %1 с %2",
        ДоговорСсылка.Номер,
        Строка(ДоговорСсылка.Контрагент));
    
    // Если не указаны согласующие, используем маршрут по умолчанию
    Если МассивСогласующих = Неопределено Тогда
        МассивСогласующих = ПолучитьМаршрутСогласованияДоговора(ДоговорСсылка);
    КонецЕсли;
    
    Результат = Обработка.ОтправитьДокументВСЭД(
        ДоговорСсылка,
        Заголовок,
        "contract",
        МассивСогласующих,
        Истина
    );
    
    Если Результат.Успех Тогда
        
        ДоговорОбъект = ДоговорСсылка.ПолучитьОбъект();
        ДоговорОбъект.ИдентификаторВСЭД = Результат.ИдентификаторДокумента;
        ДоговорОбъект.СтатусСогласования = Перечисления.СтатусыСогласования.НаСогласовании;
        ДоговорОбъект.Записать();
        
        Сообщить("Договор отправлен на согласование");
        
    Иначе
        
        ВызватьИсключение Результат.ТекстОшибки;
        
    КонецЕсли;
    
КонецПроцедуры

// Получает маршрут согласования для договора в зависимости от суммы
Функция ПолучитьМаршрутСогласованияДоговора(ДоговорСсылка)
    
    МассивСогласующих = Новый Массив;
    
    // Если сумма договора больше 1 млн - согласование у директора
    Если ДоговорСсылка.СуммаДоговора > 1000000 Тогда
        МассивСогласующих.Добавить("financial_director");
        МассивСогласующих.Добавить("director");
    Иначе
        МассивСогласующих.Добавить("manager");
    КонецЕсли;
    
    Возврат МассивСогласующих;
    
КонецФункции
```

## Синхронизация статусов

### Проверка статуса документа

```1c
Процедура ПроверитьСтатусДокументаВСЭД(ДокументСсылка)
    
    Если Не ЗначениеЗаполнено(ДокументСсылка.ИдентификаторВСЭД) Тогда
        Сообщить("Документ не был отправлен в СЭД");
        Возврат;
    КонецЕсли;
    
    Обработка = ВнешниеОбработки.Создать("ИнтеграцияДМС");
    Обработка.УстановитьПараметрыПодключения(
        Константы.АдресСерверДМС.Получить(),
        Константы.ЛогинДМС.Получить(),
        Константы.ПарольДМС.Получить()
    );
    
    Статус = Обработка.ПолучитьСтатусДокумента(ДокументСсылка.ИдентификаторВСЭД);
    
    Если Не ПустаяСтрока(Статус) Тогда
        
        СообщениеОСтатусе = "";
        
        Если Статус = "draft" Тогда
            СообщениеОСтатусе = "Черновик";
        ИначеЕсли Статус = "pending" Тогда
            СообщениеОСтатусе = "На согласовании";
        ИначеЕсли Статус = "approved" Тогда
            СообщениеОСтатусе = "Согласован";
        ИначеЕсли Статус = "rejected" Тогда
            СообщениеОСтатусе = "Отклонен";
        Иначе
            СообщениеОСтатусе = Статус;
        КонецЕсли;
        
        Сообщить(СтрШаблон("Статус документа в СЭД: %1", СообщениеОСтатусе));
        
        // Обновление статуса в 1С
        ОбновитьСтатусДокумента(ДокументСсылка, Статус);
        
    Иначе
        
        Сообщить("Не удалось получить статус документа");
        
    КонецЕсли;
    
КонецПроцедуры

Процедура ОбновитьСтатусДокумента(ДокументСсылка, СтатусСЭД)
    
    НовыйСтатус = Неопределено;
    
    Если СтатусСЭД = "approved" Тогда
        НовыйСтатус = Перечисления.СтатусыСогласования.Согласован;
    ИначеЕсли СтатусСЭД = "rejected" Тогда
        НовыйСтатус = Перечисления.СтатусыСогласования.Отклонен;
    ИначеЕсли СтатусСЭД = "pending" Тогда
        НовыйСтатус = Перечисления.СтатусыСогласования.НаСогласовании;
    КонецЕсли;
    
    Если НовыйСтатус <> Неопределено Тогда
        ДокументОбъект = ДокументСсылка.ПолучитьОбъект();
        ДокументОбъект.СтатусСогласования = НовыйСтатус;
        ДокументОбъект.Записать();
    КонецЕсли;
    
КонецПроцедуры
```

### Массовая синхронизация статусов

```1c
Процедура СинхронизироватьСтатусыДокументов()
    
    // Получаем все документы на согласовании
    Запрос = Новый Запрос;
    Запрос.Текст = 
    "ВЫБРАТЬ
    |    СчетаНаОплату.Ссылка КАК Ссылка,
    |    СчетаНаОплату.ИдентификаторВСЭД КАК ИдентификаторВСЭД
    |ИЗ
    |    Документ.СчетНаОплату КАК СчетаНаОплату
    |ГДЕ
    |    СчетаНаОплату.СтатусСогласования = ЗНАЧЕНИЕ(Перечисление.СтатусыСогласования.НаСогласовании)
    |    И СчетаНаОплату.ИдентификаторВСЭД > 0";
    
    Выборка = Запрос.Выполнить().Выбрать();
    
    Обработка = ВнешниеОбработки.Создать("ИнтеграцияДМС");
    Обработка.УстановитьПараметрыПодключения(
        Константы.АдресСерверДМС.Получить(),
        Константы.ЛогинДМС.Получить(),
        Константы.ПарольДМС.Получить()
    );
    
    КоличествоОбновлено = 0;
    
    Пока Выборка.Следующий() Цикл
        
        Статус = Обработка.ПолучитьСтатусДокумента(Выборка.ИдентификаторВСЭД);
        
        Если Не ПустаяСтрока(Статус) И (Статус = "approved" Или Статус = "rejected") Тогда
            ОбновитьСтатусДокумента(Выборка.Ссылка, Статус);
            КоличествоОбновлено = КоличествоОбновлено + 1;
        КонецЕсли;
        
    КонецЦикла;
    
    Сообщить(СтрШаблон("Обновлено статусов: %1", КоличествоОбновлено));
    
КонецПроцедуры
```

## Обработка ошибок

### Пример обработки сетевых ошибок

```1c
Функция ОтправитьДокументСПовторами(ДокументСсылка, Заголовок, ТипДокумента, МассивСогласующих)
    
    МаксимальноПопыток = 3;
    ЗадержкаМеждуПопытками = 5; // секунд
    
    Для НомерПопытки = 1 По МаксимальноПопыток Цикл
        
        Обработка = ВнешниеОбработки.Создать("ИнтеграцияДМС");
        Обработка.УстановитьПараметрыПодключения(
            Константы.АдресСерверДМС.Получить(),
            Константы.ЛогинДМС.Получить(),
            Константы.ПарольДМС.Получить()
        );
        
        Результат = Обработка.ОтправитьДокументВСЭД(
            ДокументСсылка,
            Заголовок,
            ТипДокумента,
            МассивСогласующих,
            Истина
        );
        
        Если Результат.Успех Тогда
            Возврат Результат;
        КонецЕсли;
        
        // Если это не последняя попытка, делаем паузу
        Если НомерПопытки < МаксимальноПопыток Тогда
            Сообщить(СтрШаблон("Попытка %1 не удалась. Повтор через %2 сек...", 
                НомерПопытки, ЗадержкаМеждуПопытками));
            Приостановить(ЗадержкаМеждуПопытками * 1000);
        КонецЕсли;
        
    КонецЦикла;
    
    // Все попытки исчерпаны
    Результат.ТекстОшибки = "Не удалось отправить документ после " 
        + Формат(МаксимальноПопыток, "ЧГ=") + " попыток: " + Результат.ТекстОшибки;
    
    Возврат Результат;
    
КонецФункции
```

## Дополнительные советы

### Оптимизация производительности

1. **Кэширование токенов**: Сохраняйте JWT токен в переменной сеанса, чтобы не запрашивать его при каждом вызове
2. **Пакетная обработка**: Отправляйте документы пакетами при массовой выгрузке
3. **Асинхронная обработка**: Используйте фоновые задания для отправки больших объемов данных

### Безопасность
